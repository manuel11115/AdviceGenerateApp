# Ejercicio 2 - Agenda digital (agregar, buscar, eliminar, listar)
agenda = {"Ana": "4771234567", "Luis": "4778901234"}


def agregar():
    nombre = input("Nombre a agregar: ")
    agenda[nombre] = input("Teléfono: ")
    print(f"Contacto {nombre} agregado.")


def buscar():
    nombre = input("Nombre a buscar: ")
    if nombre in agenda:
        print(f"Teléfono: {agenda[nombre]}")
    else:
        print("No encontrado")


def eliminar():
    nombre = input("Nombre a eliminar: ")
    if nombre in agenda:
        del agenda[nombre]
        print(f"Contacto {nombre} eliminado.")
    else:
        print("No encontrado")


def listar():
    for nombre, tel in agenda.items():
        print(f"{nombre}: {tel}")


acciones = {"1": agregar, "2": buscar, "3": eliminar, "4": listar}

while True:
    print("\n--- AGENDA ---\n1) Agregar  2) Buscar  3) Eliminar  4) Listar  5) Salir")
    op = input("Opción: ")
    if op == "5":
        break
    acciones.get(op, lambda: print("Opción inválida"))()

# Ejercicio 3 - Control de inventario
inventario = [
    {"producto": "Croquetas", "precio": 250, "stock": 20},
    {"producto": "Juguete", "precio": 100, "stock": 15},
    {"producto": "Shampoo", "precio": 180, "stock": 10},
]

# Calcular ingresos potenciales
total = 0
for item in inventario:
    subtotal = item["precio"] * item["stock"]
    print(f"{item['producto']}: {item['precio']} x {item['stock']} = {subtotal}")
    total += subtotal

print("Ingreso total si se vende todo:", total)

Corrida

Croquetas: 250 x 20 = 5000
Juguete: 100 x 15 = 1500
Shampoo: 180 x 10 = 1800
Ingreso total si se vende todo: 8300

# Struct en Python (4) - modulo struct (manejo binario)
import struct

# Empaquetar datos (entero y flotante) en formato binario
data = struct.pack('if', 25, 3.14)   # i = int, f = float
print("Datos empaquetados:", data)

# Desempaquetar los datos
valor = struct.unpack('if', data)
print("Datos originales:", valor)

Ejemplo práctico combinado (veterinaria)

# Ejemplo practico combinado - Veterinaria con dataclass
from dataclasses import dataclass


@dataclass
class Mascota:
    nombre: str
    especie: str
    edad: int
    vacuna: str
    costo: float


registro = [
    Mascota("Firulais", "Perro", 3, "Rabia", 250.0),
    Mascota("Michi", "Gato", 2, "Triple Felina", 300.0),
    Mascota("Max", "Perro", 1, "Parvovirus", 220.0),
]

total = 0
for m in registro:
    print(f"{m.nombre} ({m.especie}) - Vacuna: {m.vacuna} - ${m.costo}")
    total += m.costo
print("\nTotal ingresos:", total)

Corrida

Firulais (Perro) - Vacuna: Rabia - $250.0
Michi (Gato) - Vacuna: Triple Felina - $300.0
Max (Perro) - Vacuna: Parvovirus - $220.0

Total ingresos: 770.0

# Ejercicio integrador final - Gestion de una veterinaria
# Usa listas y diccionarios combinados.
PRECIOS = {"Rabia": 250.0, "Triple Felina": 300.0, "Parvovirus": 220.0, "Moquillo": 180.0}

mascotas = []  # lista de diccionarios


def registrar():
    nombre = input("Nombre: ").strip()
    tipo = input("Tipo (perro/gato/...): ").strip()
    edad = int(input("Edad: "))
    print("Vacunas disponibles:", ", ".join(f"{v} (${p})" for v, p in PRECIOS.items()))
    vacuna = input("Vacuna aplicada: ").strip()
    if vacuna not in PRECIOS:
        print("Vacuna no válida, no se registró.")
        return
    mascotas.append({"nombre": nombre, "tipo": tipo, "edad": edad, "vacuna": vacuna})
    print(f"Mascota {nombre} registrada.")


def buscar():
    nombre = input("Nombre a buscar: ").strip().lower()
    encontradas = [m for m in mascotas if m["nombre"].lower() == nombre]
    if not encontradas:
        print("No encontrada")
    for m in encontradas:
        print(f"{m['nombre']} - {m['tipo']} - {m['edad']} años - vacuna: {m['vacuna']}")


def estadisticas():
    conteo = {}
    for m in mascotas:
        conteo[m["vacuna"]] = conteo.get(m["vacuna"], 0) + 1
    print("Vacunas aplicadas:")
    for vacuna, n in conteo.items():
        print(f"  {vacuna}: {n}")
    print("Total de vacunas:", sum(conteo.values()))


def ingreso_total():
    desglose = {}
    for m in mascotas:
        desglose[m["vacuna"]] = desglose.get(m["vacuna"], 0) + PRECIOS[m["vacuna"]]
    for vacuna, monto in desglose.items():
        print(f"  {vacuna}: ${monto}")
    print("Ingreso total:", sum(desglose.values()))


acciones = {"1": registrar, "2": buscar, "3": estadisticas, "4": ingreso_total}

while True:
    print("\n--- VETERINARIA ---\n1) Registrar  2) Buscar  3) Estadísticas  4) Ingreso total  5) Salir")
    op = input("Opción: ")
    if op == "5":
        break
    acciones.get(op, lambda: print("Opción inválida"))()
